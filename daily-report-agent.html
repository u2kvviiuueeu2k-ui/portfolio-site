03 SCRIPT — 実際の自動化コード（GAS）

実運用では下記のGoogle Apps Script が毎朝8時に自動実行されます。
スプレッドシートを読み込み → Claude APIで生成 → Slackへ投稿まで全自動。

Google Apps Script — daily_report.gs 
// ================================================================
// 日次生産レポート自動生成スクリプト
// 毎朝8:00にトリガー実行 → Claude API → Slack送信
// ================================================================

const CLAUDE_API_KEY  = PropertiesService.getScriptProperties().getProperty('CLAUDE_API_KEY');
const SLACK_WEBHOOK   = PropertiesService.getScriptProperties().getProperty('SLACK_WEBHOOK_URL');
const SHEET_ID        = PropertiesService.getScriptProperties().getProperty('SPREADSHEET_ID');

function generateDailyReport() {
  // ① スプレッドシートからデータ取得
  const ss          = SpreadsheetApp.openById(SHEET_ID);
  const tankSheet   = ss.getSheetByName('タンク管理');
  const shippingSheet = ss.getSheetByName('出荷在庫');

  const tankData     = getSheetData(tankSheet);
  const shippingData = getSheetData(shippingSheet);
  const today        = new Date().toLocaleDateString('ja-JP');

  // ② Claude APIへリクエスト
  const prompt = `以下は${today}の酒造工場の日次データです。
タンク管理：${JSON.stringify(tankData)}
出荷在庫：${JSON.stringify(shippingData)}

醸造長向けの日次レポートを作成してください。
・発酵異常・温度異常のアラートがあれば最優先で記載
・当日の出荷実績と在庫サマリー
・翌日の注意事項
300文字以内で簡潔に。`;

  const response = UrlFetchApp.fetch('https://api.anthropic.com/v1/messages', {
    method:  'post',
    headers: {
      'x-api-key':         CLAUDE_API_KEY,
      'anthropic-version': '2023-06-01',
      'content-type':      'application/json'
    },
    payload: JSON.stringify({
      model:      'claude-sonnet-4-20250514',
      max_tokens: 1000,
      messages:   [{ role: 'user', content: prompt }]
    })
  });

  const report = JSON.parse(response.getContentText())
                     .content[0].text;

  // ③ Slackへ送信
  UrlFetchApp.fetch(SLACK_WEBHOOK, {
    method:  'post',
    payload: JSON.stringify({
      text: `*📊 日次レポート ${today}*\n\n` + report
    })
  });

  // ④ ログをスプレッドシートに保存
  const logSheet = ss.getSheetByName('送信ログ');
  logSheet.appendRow([new Date(), 'Slack', '送信完了', report]);
}

function getSheetData(sheet) {
  const data    = sheet.getDataRange().getValues();
  const headers = data.shift();
  return data.map(row => Object.fromEntries(
    headers.map((h, i) => [h, row[i]])
  ));
}
